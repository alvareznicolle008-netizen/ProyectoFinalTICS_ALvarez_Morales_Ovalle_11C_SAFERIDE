Problemática IRTRA Petapa

Contexto y Justificación del Proyecto

En el parque de atracciones IRTRA Petapa, el flujo constante de visitantes en juegos de alto impacto exige un control riguroso de las zonas de peligro. Los errores humanos de aproximación a las estructuras mecánicas mientras están en movimiento representan un riesgo crítico de seguridad.

SAFE RIDE resuelve esta problemática automatizando la detección de invasión de perímetro mediante sensores ultrasónicos, emitiendo alertas inmediatas antes de que ocurra una colisión o accidente.

Solución Tecnológica

Monitoreo de Proximidad y Control de Zonas

🚨

Zona de Peligro

Si un visitante u objeto se aproxima a la atracción a una distancia igual o menor a 15 cm.

Respuesta: LED Rojo Encendido + Alarma Sonora de Buzzer.
⚠️

Zona de Precaución

Si la aproximación se encuentra en un rango intermedio entre 16 cm y 50 cm.

Respuesta: LED Amarillo Parpadeante (Advertencia).
🟢

Zona Segura

Distancia mayor a 50 cm del área restringida de la atracción.

Respuesta: Indicadores en reposo / Operación normal.

// Librerias
// Instancar obj
// Definicion de PINES
// PROYECTO FINAL TECNOLOGÍA APLICADA - SAFE RIDE (IRTRA PETAPA)
//Catherine, Kimberly y Paula
const int TRIG_PIN = 4;
const int ECHO_PIN = 5;

const int LED_AMARILLO = 2; // Renombrado a amarillo para coincidir con tu circuito
const int LED_VERDE = 3;
const int LED_ROJO = 9;

const int BUZZER_PIN = 10;
const int LDR_PIN = A0;

const int DISTANCIA_PELIGRO_CM = 15;      // Distancia para LED Rojo y Buzzer
const int DISTANCIA_PRECAUCION_CM = 50;   // Distancia para que parpadee el LED Amarillo
const int LDR_UMBRAL_ARNES = 500;

// Variables para controlar el parpadeo sin detener el programa
unsigned long tiempoAnterior = 0;
const long intervaloParpadeo = 250;       // Velocidad del parpadeo en milisegundos
int estadoLedAmarillo = LOW;

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  pinMode(LED_AMARILLO, OUTPUT);
  pinMode(LED_VERDE, OUTPUT);
  pinMode(LED_ROJO, OUTPUT);

  pinMode(BUZZER_PIN, OUTPUT);

  Serial.begin(9600);
}

long medirDistanciaCM() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);

  long duracion = pulseIn(ECHO_PIN, HIGH, 30000);

  if (duracion == 0) {
    return 999;
  }

  return duracion * 0.034 / 2;
}

void loop() {
  int ldrValor = analogRead(LDR_PIN);
  long distancia = medirDistanciaCM();

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.print(" cm | LDR: ");
  Serial.println(ldrValor);

  // Guardamos el tiempo actual para calcular cuándo debe parpadear
  unsigned long tiempoActual = millis();

  // 🔴 PERSONA/OBJETO DEMASIADO CERCA (15 cm o menos)
  if (distancia <= DISTANCIA_PELIGRO_CM) {
    digitalWrite(LED_ROJO, HIGH);
    digitalWrite(LED_AMARILLO, LOW); // Apagamos el amarillo para que destaque el peligro
    tone(BUZZER_PIN, 1000);
  } 
  // 🟡 ZONA DE PRECAUCIÓN (entre 16 cm y 50 cm)
  else if (distancia <= DISTANCIA_PRECAUCION_CM) {
    digitalWrite(LED_ROJO, LOW);
    noTone(BUZZER_PIN);
    
    // Lógica para hacer parpadear el LED amarillo sin usar delay()
    if (tiempoActual - tiempoAnterior >= intervaloParpadeo) {
      tiempoAnterior = tiempoActual; // Actualizamos el temporizador
      
      // Invertimos el estado del LED (si estaba apagado lo prendemos, y viceversa)
      if (estadoLedAmarillo == LOW) {
        estadoLedAmarillo = HIGH;
      } else {
        estadoLedAmarillo = LOW;
      }
      digitalWrite(LED_AMARILLO, estadoLedAmarillo);
    }
  } 
  // 🟢 FUERA DE RANGO (> 50 cm) - Todo apagado
  else {
    digitalWrite(LED_ROJO, LOW);
    digitalWrite(LED_AMARILLO, LOW);
    noTone(BUZZER_PIN);
  }

  //  ARNÉS CONECTADO / DETECTA LUZ
  if (ldrValor > LDR_UMBRAL_ARNES) {
    digitalWrite(LED_VERDE, HIGH);
  } else {
    digitalWrite(LED_VERDE, LOW);
  }

  delay(100);
}
